// MainActivity.java
package com.example.citydash;

import android.os.Bundle;
import android.view.MotionEvent;
import android.widget.ImageView;
import android.widget.RelativeLayout;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;
import java.util.Random;

public class MainActivity extends AppCompatActivity {

    ImageView player, obstacle, coin;
    TextView scoreText;
    int score = 0;
    int playerLane = 1; // 0=left, 1=center, 2=right
    Random random = new Random();

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        
        player = findViewById(R.id.player);
        obstacle = findViewById(R.id.obstacle);
        coin = findViewById(R.id.coin);
        scoreText = findViewById(R.id.scoreText);

        // Start game loop
        player.postDelayed(gameLoop, 100);
    }

    // Touch controls
    @Override
    public boolean onTouchEvent(MotionEvent event) {
        if(event.getAction() == MotionEvent.ACTION_UP){
            float x = event.getX();
            if(x < player.getX()) moveLeft();
            else moveRight();
        }
        return super.onTouchEvent(event);
    }

    void moveLeft(){
        if(playerLane > 0){
            playerLane--;
            player.setX(playerLane * 300); // adjust lane width
        }
    }

    void moveRight(){
        if(playerLane < 2){
            playerLane++;
            player.setX(playerLane * 300);
        }
    }

    Runnable gameLoop = new Runnable() {
        @Override
        public void run() {
            // Move obstacle down
            obstacle.setY(obstacle.getY() + 20);
            coin.setY(coin.getY() + 20);

            // Collision detection
            if(Math.abs(player.getX() - obstacle.getX()) < 50 &&
               Math.abs(player.getY() - obstacle.getY()) < 50){
                // Game over
                scoreText.setText("Game Over! Score: " + score);
                return;
            }

            // Coin collect
            if(Math.abs(player.getX() - coin.getX()) < 50 &&
               Math.abs(player.getY() - coin.getY()) < 50){
                score += 10;
                scoreText.setText("Score: " + score);
                resetCoin();
            }

            // Reset obstacle if out of screen
            if(obstacle.getY() > 2000){
                obstacle.setY(-200);
                obstacle.setX(random.nextInt(3) * 300);
            }

            player.postDelayed(this, 50); // repeat loop
        }
    };

    void resetCoin(){
        coin.setY(-200);
        coin.setX(random.nextInt(3) * 300);
    }
}
